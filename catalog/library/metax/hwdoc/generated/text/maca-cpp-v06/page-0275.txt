2.18.11 __builtin_mxc_barrier
描述
屏障与同步：等待共享内存中所有写入完成。根据程序访问时间点自动插入内存栅栏。这是一个较为宽
松的屏障，建议优先使用 __syncthreads。
语法
void __builtin_mxc_barrier();
示例
__global__ void test_swap_warp_data(int *data) {
__shared__ int sharedData[128];
int id = threadIdx.x + blockIdx.x * blockDim.x;
int warpId = id / 64;
int laneId = threadIdx.x % 64;
sharedData[id] = data[id];
__builtin_mxc_barrier(); // 第一次同步，保证 sharedData 写入完毕
int targetWarpId = warpId ^ 1;
int targetId = targetWarpId * 64 + laneId;
int val = sharedData[targetId];
__builtin_mxc_barrier(); // 第二次同步，保证数据交换后再写回
data[id] = val;
}
